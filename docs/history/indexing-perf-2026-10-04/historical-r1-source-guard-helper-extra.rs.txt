
#[cfg(test)]
pub(crate) fn historical_owned_path_for_guard(label:&str)->std::path::PathBuf {
    static NEXT:AtomicU64=AtomicU64::new(1);
    let path=std::env::temp_dir().join(format!("flistwalker-historical-{label}-{}-{}",std::process::id(),NEXT.fetch_add(1,Ordering::Relaxed)));
    std::fs::create_dir(&path).expect("exclusive owned writer guard root");path
}
#[cfg(test)]
impl WriterObserver {
    pub(crate) fn physically_joined_for_guard(&self)->bool {
        let p=self.cell.lock().expect("historical cleanup observation");p.writers.get(&self.path).is_some_and(|w|w.first_wait.as_ref().is_some_and(|(ok,_)|*ok) && w.physical_join.as_ref().is_some_and(|(ok,_)|*ok))
    }
}
