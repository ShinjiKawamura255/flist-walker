"""One unauthenticated fixed-URL read per verified public asset; retain failures."""
import sys
if not __debug__ or sys.flags.optimize:
    raise RuntimeError('release observer requires Python assertions enabled; optimization is forbidden')
import concurrent.futures
import datetime
import hashlib
import importlib.util
import json
from pathlib import Path
import subprocess
import time


def receive(asset, destination):
    path = destination / asset['name']
    stderr = destination / (asset['name'] + '.download-stderr')
    receipt = destination / (asset['name'] + '.transport.json')
    if any(p.exists() for p in (path, stderr, receipt)):
        raise RuntimeError('refuse duplicate download or overwrite')
    record = dict(id=asset['id'], name=asset['name'], url=asset['browser_download_url'],
                  expected_bytes=asset['size'], expected_digest=asset['digest'],
                  status='FAIL', unauthenticated=True, attempts=1,
                  curl_max_seconds=600, watchdog_seconds=630)
    try:
        started = time.monotonic()
        with path.open('xb') as out, stderr.open('xb') as errors:
            command = ['curl', '--disable', '--proto', '=https', '--proto-redir', '=https',
                       '--location', '--fail', '--silent', '--show-error', '--max-redirs', '5',
                       '--connect-timeout', '30', '--max-time', '600',
                       asset['browser_download_url']]
            # --disable is first: no user curlrc, token lookup, cookies or credentials.
            child = subprocess.Popen(command, stdout=out, stderr=errors)
            record['owned_pid'] = child.pid
            try:
                code = child.wait(timeout=630)
            except subprocess.TimeoutExpired:
                child.kill()
                child.wait(timeout=30)
                raise TimeoutError('public asset read exceeded bounded watchdog; no automatic retry')
        record['elapsed_seconds'] = time.monotonic() - started
        record['exit_code'] = code
        raw = path.read_bytes()
        record['observed_bytes'] = len(raw)
        record['observed_digest'] = 'sha256:' + hashlib.sha256(raw).hexdigest()
        if code != 0 or len(raw) != asset['size'] or record['observed_digest'] != asset['digest']:
            raise RuntimeError('public download exit/size/digest mismatch; original received bytes retained')
        record['status'] = 'PASS_TRANSPORT_AND_API_DIGEST'
    except Exception as error:
        record['error'] = f'{type(error).__name__}: {error}'
        if path.exists():
            record['retained_bytes'] = path.stat().st_size
        raise
    finally:
        with receipt.open('x') as output:
            json.dump(record, output, indent=2)
            output.write('\n')
    return record


def main():
    import argparse
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('metadata', type=Path)
    parser.add_argument('destination', type=Path)
    parser.add_argument('--release-id', required=True, type=int)
    args = parser.parse_args()
    release = json.loads(args.metadata.read_text())
    assert release['id'] == args.release_id and release['tag_name'] == release['name'] == 'v0.32.0'
    assert release['target_commitish'] == '77e36c4eef21b5047747d04b3983835a50d92f06'
    assert not release['draft'] and not release['prerelease'] and release['published_at']
    spec = importlib.util.spec_from_file_location('bundle', Path(__file__).with_name('verify_bundle.py'))
    bundle = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(bundle)
    assets = release['assets']
    expected = bundle.inventory('0.32.0') | {'SHA256SUMS', 'SHA256SUMS.sig'}
    assert len(assets) == len({a['id'] for a in assets}) == len(expected) == 28
    assert {a['name'] for a in assets} == expected
    for asset in assets:
        assert asset['state'] == 'uploaded' and asset['size'] > 0
        assert asset['browser_download_url'] == ('https://github.com/ShinjiKawamura255/flist-walker/releases/download/v0.32.0/' + asset['name'])
        assert asset['digest'].startswith('sha256:') and len(asset['digest']) == 71
    assert not args.destination.exists()
    args.destination.mkdir()
    # Transport receipts live separately from the exact 28-member bundle.
    received = args.destination.with_name(args.destination.name + '-received')
    assert not received.exists()
    received.mkdir()
    with concurrent.futures.ThreadPoolExecutor(max_workers=4) as pool:
        records = list(pool.map(lambda a: receive(a, received), assets))
    for asset in assets:
        (args.destination / asset['name']).write_bytes((received / asset['name']).read_bytes())
    report = dict(utc=datetime.datetime.now(datetime.timezone.utc).isoformat(),
                  release_id=release['id'], tag=release['tag_name'], source=release['target_commitish'],
                  unauthenticated_public_browser_urls=True, attempts_per_asset=1,
                  assets=records, result='PASS_ALL_28_PUBLIC_DOWNLOADS')
    out = args.destination.with_name(args.destination.name + '-transport.json')
    with out.open('x') as stream:
        json.dump(report, stream, indent=2)
        stream.write('\n')
    print(report['result'])


if __name__ == '__main__':
    main()
